# Bug: the "Choose interaction" menu didn't say where each choice is, so with two
# similar things nearby you couldn't tell which one you'd use. The menu is now a 3x3 grid
# of the tiles around the player, and each choice in the list starts with the key for
# its direction.
seed 12345
size 100x30
skip-splash
key ui.MENU_PREV
# Next to the corpse Game::new leaves at (18, 21, 0), with a chest placed to the west
cheat tp 17 21 0
cheat giveitem chest
key build.TOGGLE_BREAK_PLACE_MODE 2
key hotbar.HOTBAR_SLOT_1 2
# The picker lists Clear slot, then placeable items in inventory order: Acorn, Chest
key movement.MOVE_SOUTH 2
key ui.MENU_ACTIVATE
key build.PLACE_WEST
key build.TOGGLE_BREAK_PLACE_MODE 2
key action.INTERACT
expect-text "[6] Loot corpse (2 items)"
expect-text "[4] Open chest"
expect-text "Direction keys: pick a tile"
screen choose-interaction
# Nothing north: the menu stays open
key movement.MOVE_NORTH
expect-text "Nothing there"
# West opens the chest at once
key movement.MOVE_WEST
expect-text "Opened chest"
